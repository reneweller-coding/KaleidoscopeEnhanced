# ---------------------------------------------------------------------------
# CMakeLists.txt -- the Linux/macOS build.
#
# The WINDOWS build is Kaleidoscope.vcxproj and stays that way. That project is
# the reference: it is what the releases are built with, what deploy.ps1 drives,
# and what every visual regression has been measured against. This file exists
# alongside it, never instead of it, and it refuses to configure on Windows so
# the two can never quietly disagree about how the shipped binary is produced.
#
# Build:
#   cmake -S . -B build -DCMAKE_BUILD_TYPE=Release
#   cmake --build build -j
#
# The executable lands in build/, but the program resolves its assets relative
# to the CURRENT DIRECTORY as "../Engine/...", "../Scene2D/..." and so on -- the
# same convention the Windows package uses, where the exe sits in bin/. So run
# it from a subdirectory of the repo, e.g.:
#   cd build && ./Kaleidoscope
# ---------------------------------------------------------------------------
cmake_minimum_required(VERSION 3.16)
project(Kaleidoscope LANGUAGES CXX)

if(WIN32)
	message(FATAL_ERROR
		"On Windows, build Kaleidoscope.vcxproj -- that project is the reference "
		"build and this CMake file is deliberately not a second way to produce it.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
	set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# AUTOMOC rather than the checked-in moc/ folder: those files are generated by
# the Windows Qt's moc and are tied to its version, so they are left to the
# vcxproj and regenerated here for whatever Qt is installed.
set(CMAKE_AUTOMOC ON)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Xml Multimedia OpenGL OpenGLWidgets)
find_package(OpenGL REQUIRED)

# ---------------------------------------------------------------------------
# Sources: everything in Source/, minus the Windows-only translation units.
#
# Listed explicitly rather than globbed, so that adding a file is a deliberate
# act on both build systems and the two lists can be diffed against each other.
# ---------------------------------------------------------------------------
set(KALEIDO_SOURCES
	Source/AudioAnalyzer.cpp
	Source/AudioConditioner.cpp
	Source/ComputeFX.cpp
	Source/Configuration.cpp
	Source/EffectShader.cpp
	Source/ExprEval.cpp
	Source/GpuSims.cpp
	Source/MeshImport.cpp
	Source/MidiInput.cpp
	Source/NowPlaying.cpp
	Source/OscSender.cpp
	Source/Platform.cpp
	Source/PresentPass.cpp
	Source/QMyWindow.cpp
	Source/Recorder.cpp
	Source/RenderPipeline.cpp
	Source/Scene3DShader.cpp
	Source/SceneScheduler.cpp
	Source/SpoutIn.cpp
	Source/SpoutOut.cpp
	Source/Strings.cpp
	Source/TextureEffectKaleidoscopeBase.cpp
	Source/TrackMedia.cpp
	Source/Uniform.cpp
	Source/UpdateCheck.cpp
	Source/Utils.cpp
	Source/Vector3D.cpp
	Source/VideoIn.cpp
	Source/VideoPiP.cpp
	Source/WebRemote.cpp
	Source/glcore.cpp
	Source/glwidget.cpp
	Source/main.cpp
	Source/mesh.cpp
	Source/shader_setup.cpp
	Source/textfile.cpp
)
# Headers are listed only so AUTOMOC scans them for Q_OBJECT.
file(GLOB KALEIDO_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/Source/*.h")

add_executable(Kaleidoscope ${KALEIDO_SOURCES} ${KALEIDO_HEADERS})

target_include_directories(Kaleidoscope PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}
	# Source/ is deliberately NOT here, although the vcxproj lists it. Every
	# project header is included with quotes, so it resolves relative to the
	# including file and needs no -I. Adding it would put Source/Strings.h in
	# reach of `#include <strings.h>` on a case-INSENSITIVE filesystem (a
	# Windows drive under WSL, or macOS): glibc's string.h pulls that in from
	# inside an extern "C" block, so every caller of Strings::T() would then
	# link against an unmangled `T` that nothing defines.
	${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty
	# Case bridge for <GL/GLU.h>; see Compat/GL/GLU.h. Listed LAST so it can
	# only ever be reached for a header the real include path does not have.
	${CMAKE_CURRENT_SOURCE_DIR}/Compat
)
target_compile_definitions(Kaleidoscope PRIVATE _CRT_SECURE_NO_WARNINGS)

target_link_libraries(Kaleidoscope PRIVATE
	Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Xml
	Qt6::Multimedia Qt6::OpenGL Qt6::OpenGLWidgets
	OpenGL::GL
	OpenGL::GLU   # gluErrorString / gluPerspective
)
if(NOT APPLE)
	# glcore resolves its entry points through glXGetProcAddress, with dlsym as
	# the fallback for implementations that lack it.
	find_package(X11)
	target_link_libraries(Kaleidoscope PRIVATE ${CMAKE_DL_LIBS})
	if(OpenGL_GLX_FOUND)
		target_link_libraries(Kaleidoscope PRIVATE OpenGL::GLX)
	endif()
else()
	target_link_libraries(Kaleidoscope PRIVATE ${CMAKE_DL_LIBS})
endif()

# ---------------------------------------------------------------------------
# Loopback audio capture.
#
# WASAPI's loopback mode has one portable counterpart in practice: recording a
# PulseAudio sink's MONITOR source, which PipeWire's pulse server also serves.
# It is REQUIRED on Linux -- a music visualiser that cannot hear the music is
# not the program -- but only optional on macOS, where PulseAudio is a homebrew
# extra and the native answer (CoreAudio) cannot tap system output at all
# without a virtual device. Without it the build still produces a working
# program for offline analysis (-w / -x).
# ---------------------------------------------------------------------------
find_package(PkgConfig)
if(PkgConfig_FOUND)
	pkg_check_modules(PULSE libpulse-simple)
endif()
if(PULSE_FOUND)
	target_compile_definitions(Kaleidoscope PRIVATE KALEIDO_HAVE_PULSE)
	target_include_directories(Kaleidoscope PRIVATE ${PULSE_INCLUDE_DIRS})
	target_link_libraries(Kaleidoscope PRIVATE ${PULSE_LIBRARIES})
	message(STATUS "Kaleidoscope: loopback capture via PulseAudio")
elseif(APPLE)
	message(WARNING "Kaleidoscope: no libpulse-simple -- live capture disabled, "
	                "offline analysis (-w/-x) still works.")
else()
	message(FATAL_ERROR "Kaleidoscope: libpulse-simple not found. Install it "
	                    "(Debian/Ubuntu: libpulse-dev) -- without it the visualiser "
	                    "cannot hear the system audio.")
endif()
