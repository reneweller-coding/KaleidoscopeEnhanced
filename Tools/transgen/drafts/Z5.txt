about: A triangulated grid of facets turns over from the centre outwards, each facet shading like paper at its fold angle; on the back of every facet lies the incoming scene.
music: audioSwell -> a slight breath of the brightness in the middle (light only)
knob: facetP 0.5 2.2 the number of facets
knob: creaseP 0.5 2.0 how marked the shading of the folds is

huegain: 0.0
---
vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float asp = resolution.x / resolution.y;
    float n = ( 5.0 + 4.0 * gKA );
    vec2  q = vec2( p.x * asp, p.y ) * n;
    vec2  s = vec2( q.x + q.y * 0.57735027, q.y * 1.15470054 );
    vec2  g = floor( s );
    vec2  fr = s - g;
    float up = ( fr.x + fr.y > 1.0 ) ? 1.0 : 0.0;
    vec2  id = g + up * 0.5;
    float order = clamp( length( vec2( ( id.x - q.y * 0.0 ) / n / asp * 1.0, id.y / n ) - vec2( 0.5 ) ) * 0.9 + 0.25 * hash21( id + 3.3 ), 0.0, 1.0 );
    float u = clamp( d * 1.5 - 0.5 * order, 0.0, 1.0 );
    float w = smoothstep( 0.45, 0.55, u );
    float shade = 1.0 - 0.28 * gKB * sin( Pi * u ) * ( 0.5 + 0.5 * hash21( id + 7.7 ) ) * 1.6;
    vec4  c = mix( a, b, w );
    c.rgb *= shade;
    return c;
}

/**  The hexagonal cell of a point: the point's offset from the cell's centre and the centre itself.
 *
 * @param p  The point in cell units. (in)
 *
 * @return
 *   xy: the offset from the centre, zw: the centre.
 **/
vec4 hexInfo( vec2 p )
{
    vec2 r = vec2( 1.0, 1.7320508 );
    vec2 h = r * 0.5;
    vec2 a = mod( p, r ) - h;
    vec2 b = mod( p - h, r ) - h;
    vec2 g = ( dot( a, a ) < dot( b, b ) ) ? a : b;
    return vec4( g, p - g );
}

vec4 trans( vec2 p, float d )
{
    float asp = resolution.x / resolution.y;
    float n = ( 5.0 + 4.0 * gKA );
    vec2  q = vec2( p.x * asp, p.y ) * n;
    vec4  hx = hexInfo( q );
    vec2  c = hx.zw;
    vec2  pc = vec2( c.x / n / asp, c.y / n );
    float order = clamp( length( pc - vec2( 0.5 ) ) * 1.1 + 0.3 * gKB * 0.5 * hash21( c + 2.9 ), 0.0, 1.0 );
    float u = clamp( d * 1.5 - 0.5 * order, 0.0, 1.0 );
    if ( u <= 0.0 )
        return tA( p );
    if ( u >= 1.0 )
        return tB( p );
    float th = Pi * smoothstep( 0.0, 1.0, u );
    float ct = cos( th );
    float sx = max( abs( ct ), 0.04 );
    vec2  l = hx.xy;
    vec2  uvc = pc + vec2( l.x / sx, l.y ) / n / vec2( asp, 1.0 );
    float inside = ( abs( l.x / sx ) < 0.5 ) ? 1.0 : 0.0;
    vec4  col = ( ct >= 0.0 ) ? tAm( uvc ) : tBm( uvc );
    col.rgb *= 0.55 + 0.45 * sx;
    return mix( vec4( mix( tA( p ), tB( p ), step( 0.5, u ) ).rgb * 0.45, 1.0 ), col, inside );
}

=== ParquetSlideInterlock
brief: TRANSITION PARQUET SLIDE INTERLOCK: strips of parquet slide into one another from alternating sides and change the scene as they pass.
about: The picture is cut into horizontal rows of planks; each row slides across in the opposite direction to its neighbours, at its own time, and the new scene is behind each sliding edge.
music: audioSwell -> a slight breath of the brightness in the middle (light only)
knob: rowsP 0.5 2.2 the number of rows
knob: staggerP 0.5 2.0 how far apart the rows start
moods: wipe,blocky

cx: 1
huegain: 0.0
---
vec4 trans( vec2 p, float d )
{
    float n = floor( 6.0 + 5.0 * gKA );
    float row = floor( p.y * n );
    float dir = ( mod( row, 2.0 ) < 0.5 ) ? 1.0 : -1.0;
    float L = 0.5 * gKB;
    float ph = clamp( d * ( 1.0 + L ) - L * hash21( vec2( row, 8.0 ) ), 0.0, 1.0 );
    ph = smoothstep( 0.0, 1.0, ph );
    float front = ( dir > 0.0 ) ? ph : 1.0 - ph;
    float w = ( dir > 0.0 ) ? smoothstep( front + 0.015, front - 0.015, p.x ) : smoothstep( front - 0.015, front + 0.015, p.x );
    vec2  sh = vec2( dir * ph * 0.06, 0.0 );
    vec4  a = tAm( p - sh );
    vec4  b = tBm( p + vec2( dir * ( 1.0 - ph ) * 0.06, 0.0 ) );
    float edge = exp( -pow( ( p.x - front ) / 0.01, 2.0 ) );
    vec4  c = mix( a, b, w );
    c.rgb *= 1.0 - 0.3 * edge * sin( Pi * ph );
    return c;
}

=== ConcentricRingRotateReveal
brief: TRANSITION CONCENTRIC RING ROTATE REVEAL: concentric rings turn against each other, each showing the new scene in a growing angle.

---
vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec2  r = ( p - 0.5 ) * vec2( resolution.x / resolution.y, 1.0 );
    float rad = length( r );
    float n = floor( 4.0 + 3.0 * gKA );
    float ring = floor( rad * n * 1.4 );
    float dir = ( mod( ring, 2.0 ) < 0.5 ) ? 1.0 : -1.0;
    float ang = atan( r.y, r.x ) / 6.2831853 + 0.5;
    float L = 0.4;
    float ph = clamp( d * ( 1.0 + L ) - L * hash21( vec2( ring, 3.0 ) ), 0.0, 1.0 );
    float start = fract( dir * ph * 0.5 * gKB + 0.13 * ring );
    float u = fract( ang - start );
    float w = smoothstep( ph + 0.02, ph - 0.02, u );
    w = mix( w, 1.0, step( 1.0, ph ) );
    return mix( a, b, ( ph <= 0.0 ) ? 0.0 : w );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float asp = resolution.x / resolution.y;

    vec2  q = vec2( p.x * asp, p.y ) * n;
    vec2  g = floor( q );
    vec2  l = fract( q ) - 0.5;

=== VoronoiPlateSlideApart
brief: TRANSITION VORONOI PLATE SLIDE APART: plates of the old scene shrink and slide apart, and the new scene fills the gaps between them.
about: The picture is a mosaic of irregular plates; each shrinks towards its own centre at its own time, a dark seam opens round it, and the new scene lies behind.
music: audioSwell -> a slight breath of the brightness in the middle (light only)
knob: platesP 0.5 2.2 the number of plates
knob: gapP 0.5 2.0 how dark the seams are
moods: blocky,wipe
minfade: 5
prob: 0.06
cx: 2
huegain: 0.0
---
vec4 trans( vec2 p, float d )
{
    float asp = resolution.x / resolution.y;
    float n = ( 4.0 + 3.5 * gKA );
    vec2  q = vec2( p.x * asp, p.y ) * n;
    vec2  g = floor( q );
    float f1 = 9.0;
    float f2 = 9.0;
    vec2  c1 = g;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2  c = g + vec2( float( i ), float( j ) );
            vec2  pt = c + 0.15 + 0.7 * hash22( c + 5.0 );
            float dd = length( q - pt );
            if ( dd < f1 )
            {
                f2 = f1;
                f1 = dd;
                c1 = pt;
            }
            else if ( dd < f2 )
                f2 = dd;
        }
    float t = f1 / ( f1 + f2 + 1.0e-4 ) * 2.0;
    float order = hash21( floor( c1 * 3.0 ) + 1.7 );
    float u = clamp( d * 1.5 - 0.5 * order, 0.0, 1.0 );
    float k = max( 1.0 - u, 0.05 );
    vec2  pc = c1 / n / vec2( asp, 1.0 );
    vec2  l = p - pc;
    vec4  a = tAm( pc + l / k );
    float inside = 1.0 - smoothstep( k - 0.04, k, t );
    float seam = smoothstep( k + 0.05, k, t ) * ( 1.0 - inside );
    vec4  c = mix( tB( p ), a, inside );
    c.rgb *= 1.0 - 0.5 * gKB * seam * sin( Pi * u );
    return c;
}
