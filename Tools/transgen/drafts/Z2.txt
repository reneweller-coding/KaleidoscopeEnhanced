/**  The nearest drop of one layer of drops: its centre and its random size.
 *
 * @param q      The place in layer units. (in)
 * @param layer  The layer, whose drops are placed differently. (in)
 * @param rnd    The drop's random number 0..1. (out)
 *
 * @return
 *   The centre of the nearest drop, in layer units.
 **/
vec2 dropNearest( vec2 q, float layer, out float rnd )
{
    vec2  g = floor( q );
    float best = 9.0;
    vec2  bc = g;
    rnd = 0.0;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2  c = g + vec2( float( i ), float( j ) );
            vec2  pt = c + 0.2 + 0.6 * hash22( c + layer * 5.7 );
            float dd = length( q - pt );
            if ( dd < best )
            {
                best = dd;
                bc = pt;
                rnd = hash21( c + layer * 9.1 );
            }
        }
    return bc;
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    float asp = resolution.x / resolution.y;
    float n = ( 6.0 + 5.0 * gKA );
    vec2  q = p * vec2( asp, 1.0 ) * n;
    float rnd;
    vec2  c = dropNearest( q, 0.0, rnd );
    float dist = length( q - c );
    float grow = smoothstep( 0.0, 0.85, d );
    float R = ( 0.18 + 0.28 * rnd ) * ( 0.3 + 1.5 * grow );
    float inside = 1.0 - smoothstep( R * 0.92, R, dist );
    float upright = smoothstep( 0.55, 1.0, d );
    float k = mix( -0.9 * gKB, 1.0, upright );
    vec2  pc = c / vec2( asp, 1.0 ) / n;
    vec2  uv = pc + ( p - pc ) * k;
    vec4  lens = tBm( uv );
    float rim = smoothstep( R * 0.6, R, dist ) * inside;
    lens.rgb *= 1.0 - 0.25 * rim * trMid( d );
    vec4  dull = mix( a, trBlurA( p, 4.0 * trMid( d ) ), 0.6 );
    float full = smoothstep( 0.82, 1.0, d );
    vec4  c0 = mix( dull, lens, inside );
    return mix( c0, tB( p ), full );
}

    return vec4( mix( c.rgb, ice.rgb, clamp( frost, 0.0, 0.4 ) ), c.a );
}

=== StarTwinkleDissolve
brief: TRANSITION STAR TWINKLE DISSOLVE: the old scene falls into night and the new one is born from stars, each a window on the new picture.
about: The old picture dims into a deep blue dark; points of light appear in it, each a growing round window that shows the new scene; the points join until the frame is full.
music: audioSwell -> the stars' slow shimmer
knob: starP 0.5 2.2 how many stars there are
knob: dimP 0.5 2.0 how deep the night gets
moods: dark,dreamy
minfade: 8
prob: 0.07
cx: 2
huegain: 0.05
---
vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float asp = resolution.x / resolution.y;
    float n = ( 12.0 + 9.0 * gKA );
    vec2  q = p * vec2( asp, 1.0 ) * n;
    vec2  g = floor( q );
    float best = 9.0;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2 c = g + vec2( float( i ), float( j ) );
            vec2 pt = c + 0.15 + 0.7 * hash22( c + 4.4 );
            best = min( best, length( q - pt ) );
        }
    float f = clamp( best * 1.05, 0.0, 1.0 );
    float w = trThresh( f, d, 0.1 );
    float m = trMid( d );
    float tw = 0.85 + 0.15 * sin( trClock() * 6.0 + hash21( floor( q ) ) * 6.2831853 + 2.0 * audioSwell );
    vec3  night = vec3( 0.02, 0.03, 0.09 );
    vec4  ao = vec4( mix( a.rgb, night + 0.25 * a.rgb, clamp( 0.8 * gKB * m, 0.0, 0.9 ) ), a.a );
    vec4  c = mix( ao, b, w );
    float star = exp( -best * best * 60.0 ) * ( 1.0 - w ) * m * tw;
    c.rgb += vec3( 0.9, 0.95, 1.0 ) * star * 0.7;
    return c;
}

=== TideFoamLaceworkReveal
brief: TRANSITION TIDE FOAM LACEWORK REVEAL: the foam of an ebbing wave draws a net of arcs, and the new scene shows through the meshes.
about: A cellular net of foam, thick at the cell edges and clear in the cells, advances over the picture; the new scene appears first in the clear meshes and the white edges thin out last.
music: audioSwell -> a slight breath of the brightness in the middle (light only)
knob: meshP 0.5 2.2 the size of the meshes
knob: foamP 0.5 2.0 how white the foam is
moods: calm,wipe
minfade: 7
prob: 0.07
cx: 2
huegain: 0.05
---
vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float asp = resolution.x / resolution.y;
    vec2  q = p * vec2( asp, 1.0 ) * ( 6.0 / gKA ) + vec2( 0.3 * trClock(), 0.0 );
    vec2  g = floor( q );
    float f1 = 9.0;
    float f2 = 9.0;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2  c = g + vec2( float( i ), float( j ) );
            float dd = length( q - ( c + 0.15 + 0.7 * hash22( c + 2.2 ) ) );
            if ( dd < f1 )
            {
                f2 = f1;
                f1 = dd;
            }
            else if ( dd < f2 )
                f2 = dd;
        }
    float edge = clamp( 1.0 - 3.5 * ( f2 - f1 ), 0.0, 1.0 );
    float sweep = clamp( p.x * 0.6 + p.y * 0.4, 0.0, 1.0 );
    float f = clamp( 0.65 * sweep + 0.2 * f1 + 0.35 * edge, 0.0, 1.0 );
    float w = trThresh( f, d, 0.14 );
    float m = trMid( d );
    vec4  c = mix( a, b, w );
    c.rgb = mix( c.rgb, vec3( 0.95, 0.97, 1.0 ), 0.55 * gKB * edge * m * ( 1.0 - 0.5 * w ) );
    return c;
}
