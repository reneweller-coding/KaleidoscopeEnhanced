/**  The hue of a colour as a fraction of the colour wheel.
 *
 * @param c  The colour. (in)
 *
 * @return
 *   The hue, 0..1 (0 for greys).
 **/
float hueOf( vec3 c )
{
    float mx = max( c.r, max( c.g, c.b ) );
    float mn = min( c.r, min( c.g, c.b ) );
    float dl = mx - mn;
    if ( dl < 1.0e-4 )
        return 0.0;
    float h = ( mx == c.r ) ? mod( ( c.g - c.b ) / dl, 6.0 ) : ( ( mx == c.g ) ? ( c.b - c.r ) / dl + 2.0 : ( c.r - c.g ) / dl + 4.0 );
    return fract( h / 6.0 );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float mx = max( a.r, max( a.g, a.b ) );
    float mn = min( a.r, min( a.g, a.b ) );
    float chroma = clamp( ( mx - mn ) * 6.0, 0.0, 1.0 );
    float f0 = mix( mix( clamp( 0.5 + 1.7 * ( trLuma( a.rgb ) - trMeanLumaA() ), 0.0, 1.0 ), smoothstep( 0.3, 0.7, fbm2( p * 2.0 + 3.0 ) ), 0.6 ), mix( fract( hueOf( a.rgb ) + 0.37 * gKB ), smoothstep( 0.3, 0.7, fbm2( p * 2.0 + 3.0 ) ), 0.5 ), chroma );
    float f = smoothstep( 0.0, 1.0, f0 );   // the order values pile up in the middle; this spreads them over the whole fade
    float w = trThresh( f, d, 0.13 * gKA * ( 1.0 + 0.3 * audioSwell ) );
    return mix( a, b, w );
}

/**  The strength of the edges of the outgoing scene, blurred over a few pixels.
 *
 * @param p  The place. (in)
 * @param r  The radius in pixels. (in)
 *
 * @return
 *   The edge strength, 0..1.
 **/
float edgeA( vec2 p, float r )
{
    vec2  px = r / resolution;
    float lx = trLuma( tA( p + vec2( px.x, 0.0 ) ).rgb ) - trLuma( tA( p - vec2( px.x, 0.0 ) ).rgb );
    float ly = trLuma( tA( p + vec2( 0.0, px.y ) ).rgb ) - trLuma( tA( p - vec2( 0.0, px.y ) ).rgb );
    return clamp( length( vec2( lx, ly ) ) * 3.0, 0.0, 1.0 );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float e = 0.5 * ( edgeA( p, 3.0 * gKB ) + edgeA( p, 9.0 * gKB ) );
    float f = mix( 1.0 - pow( clamp( e * ( 0.9 + 0.5 * gKA ), 0.0, 1.0 ), 0.45 ), smoothstep( 0.3, 0.7, fbm2( p * 2.0 + 5.0 ) ), 0.65 );
    float w = trThresh( f, d, 0.16 * ( 1.0 + 0.3 * audioSwell ) );
    return mix( a, b, w );
}

/**  The roughness of the outgoing scene at a place: the spread of its brightness over a small neighbourhood.
 *
 * @param p  The place. (in)
 * @param r  The radius in pixels. (in)
 *
 * @return
 *   The roughness, 0..1.
 **/
float roughA( vec2 p, float r )
{
    vec2  px = r / resolution;
    float c = trLuma( tA( p ).rgb );
    float s = abs( trLuma( tA( p + vec2( px.x, 0.0 ) ).rgb ) - c ) + abs( trLuma( tA( p - vec2( px.x, 0.0 ) ).rgb ) - c )
            + abs( trLuma( tA( p + vec2( 0.0, px.y ) ).rgb ) - c ) + abs( trLuma( tA( p - vec2( 0.0, px.y ) ).rgb ) - c );
    return clamp( s * 2.5, 0.0, 1.0 );
}

/**  The mean roughness of the outgoing scene, estimated at four fixed points (so the order of the roughness does not depend
 *  on how detailed the scene is as a whole).
 *
 * @param r  The radius in pixels. (in)
 *
 * @return
 *   The mean roughness, 0..1.
 **/
float meanRoughA( float r )
{
    return 0.25 * ( roughA( vec2( 0.3, 0.3 ), r ) + roughA( vec2( 0.7, 0.3 ), r ) + roughA( vec2( 0.3, 0.7 ), r ) + roughA( vec2( 0.7, 0.7 ), r ) );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    float rad = 4.0 * gKB;
    float fr = pow( 1.0 - exp( -roughA( p, rad ) * ( 1.0 + 0.5 * gKA ) / ( meanRoughA( rad ) + 0.03 ) ), 0.5 );   // the roughness as a share of the typical one
    float f = mix( fr, smoothstep( 0.3, 0.7, fbm2( p * 2.0 + 9.0 ) ), 0.65 );
    float dw = d + 0.3 * sin( 2.0 * Pi * d ) / ( 2.0 * Pi );   // the order values pile up in the middle: slower there, faster at both ends
    float w = trThresh( f, dw, 0.15 * ( 1.0 + 0.3 * audioSwell ) );
    return mix( a, b, w );
}

/**  The distance to the nearest speck of one layer of drifting specks.
 *
 * @param p      The place in layer units. (in)
 * @param layer  The layer, whose specks are placed differently. (in)
 *
 * @return
 *   The distance, in cells (at most about 1).
 **/
float moteDist( vec2 p, float layer )
{
    vec2  g = floor( p );
    float best = 2.0;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2 c = g + vec2( float( i ), float( j ) );
            vec2 pt = c + 0.15 + 0.7 * hash22( c + layer * 7.31 );
            best = min( best, length( p - pt ) );
        }
    return best;
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec2  q = p * vec2( resolution.x / resolution.y, 1.0 );
    float t = trClock() * gKB;
    float n = 5.0 * gKA;
    float d0 = moteDist( q * n + vec2( 0.4 * t, 0.2 * t ), 0.0 );
    float d1 = moteDist( q * n * 1.7 + vec2( -0.3 * t, 0.5 * t ), 1.0 );
    float f = smoothstep( 0.05, 0.75, 0.9 * min( d0, 1.25 * d1 ) );
    float w = trThresh( f, d, 0.22 );
    vec4  c = mix( a, b, w );
    c.rgb += vec3( 1.0, 0.92, 0.78 ) * 0.06 * trMid( d ) * ( 1.0 - smoothstep( 0.0, 0.35, f ) ) * ( 1.0 + audioSwell );
    return c;
}

/**  A noise of long thin fibres in one direction.
 *
 * @param p      The place. (in)
 * @param angle  The direction of the fibres in radians. (in)
 * @param len    The elongation: how many times longer than wide. (in)
 *
 * @return
 *   The fibre value, 0..1.
 **/
float fibre( vec2 p, float angle, float len )
{
    float c = cos( angle );
    float s = sin( angle );
    vec2  q = vec2( c * p.x + s * p.y, -s * p.x + c * p.y );
    return noise2( q * vec2( 1.0, len ) );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec2  q = p * vec2( resolution.x / resolution.y, 1.0 ) * ( 6.0 + 4.0 * gKA );
    float len = 7.0;
    float f = 0.4 * fibre( q, 0.35, len ) + 0.3 * fibre( q + 3.1, 1.9, len ) + 0.3 * mix( noise2( q * 0.7 ), fibre( q - 5.0, 2.8, len ), clamp( 0.4 * gKB, 0.0, 1.0 ) );
    f = smoothstep( 0.2, 0.8, f );
    float w = trThresh( f, d, 0.2 );
    return mix( a, b, w );
}
