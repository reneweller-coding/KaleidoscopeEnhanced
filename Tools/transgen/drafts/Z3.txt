/**  A lace pattern: threads along the edges of a cellular net with a ring motif in every mesh.
 *
 * @param q  The place in lace units. (in)
 *
 * @return
 *   The thread density, 0..1 (1 on a thread).
 **/
float laceThread( vec2 q )
{
    vec2  g = floor( q );
    float f1 = 9.0;
    float f2 = 9.0;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2  c = g + vec2( float( i ), float( j ) );
            float dd = length( q - ( c + 0.25 + 0.5 * hash22( c + 1.1 ) ) );
            if ( dd < f1 )
            {
                f2 = f1;
                f1 = dd;
            }
            else if ( dd < f2 )
                f2 = dd;
        }
    float net = 1.0 - smoothstep( 0.0, 0.06, f2 - f1 );
    float ring = 1.0 - smoothstep( 0.0, 0.035, abs( f1 - 0.26 ) );
    return clamp( max( net, 0.8 * ring ), 0.0, 1.0 );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec2  q = p * vec2( resolution.x / resolution.y, 1.0 ) * ( 12.0 / gKA );
    float thr = laceThread( q );
    float f = clamp( 0.3 * thr + 0.7 * smoothstep( 0.2, 0.8, fbm2( p * 3.0 + 2.0 ) ), 0.0, 1.0 );
    float w = trThresh( f, d, 0.25 );
    float m = trMid( d );
    vec4  c = mix( a, b, w );
    float veil = thr * m * 1.1 * gKB;
    return vec4( mix( c.rgb, vec3( 0.97, 0.95, 0.92 ), clamp( veil, 0.0, 0.45 ) ), c.a );
}

/**  A gauze weave: a thin lattice of threads at an angle.
 *
 * @param q    The place in weave units. (in)
 * @param ang  The angle of the weave in radians. (in)
 *
 * @return
 *   The thread density, 0..1.
 **/
float gauze( vec2 q, float ang )
{
    float c = cos( ang );
    float s = sin( ang );
    vec2  r = vec2( c * q.x + s * q.y, -s * q.x + c * q.y );
    vec2  th = abs( fract( r ) - 0.5 );
    return clamp( 1.0 - 4.0 * min( th.x, th.y ), 0.0, 1.0 ) * 0.6 + 0.4 * ( 1.0 - 2.0 * max( th.x, th.y ) );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec2  q = p * vec2( resolution.x / resolution.y, 1.0 ) * ( 60.0 * gKA );
    float lag = 0.16 * gKB;
    float l1 = 1.0 - smoothstep( 0.0, 0.5 - lag, d - 0.1 );
    float l2 = 1.0 - smoothstep( 0.2, 0.7, d );
    float l3 = 1.0 - smoothstep( 0.5 + lag, 1.0, d );
    float v = 0.25 * l1 * gauze( q, 0.2 ) + 0.22 * l2 * gauze( q * 0.7, 0.9 ) + 0.2 * l3 * gauze( q * 1.3, 1.6 );

}
