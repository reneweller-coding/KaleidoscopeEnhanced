/**  A disc-shaped blur of the outgoing scene that favours bright points (bokeh).
 *
 * @param p  The place. (in)
 * @param r  The radius in pixels. (in)
 *
 * @return
 *   The colour.
 **/
vec4 discA( vec2 p, float r )
{
    vec4  s = vec4( 0.0 );
    float ws = 0.0;
    for ( int i = 0; i < 16; ++i )
    {
        float a = 2.39996323 * float( i ) + 6.2831853 * hash21( gl_FragCoord.xy );
        float rr = sqrt( ( float( i ) + 0.5 ) / 16.0 );
        vec4  c = tA( p + vec2( cos( a ), sin( a ) ) * rr * r / resolution );
        float w = 1.0 + 5.0 * pow( trLuma( c.rgb ), 3.0 );
        s += c * w;
        ws += w;
    }
    return s / ws;
}

/**  The effect of a turning polariser on a colour: glare out, saturation up.
 *
 * @param c    The colour. (in)
 * @param amt  The strength of the filter, 0..1. (in)
 *
 * @return
 *   The filtered colour.
 **/
vec3 polarise( vec3 c, float amt )
{
    float l = trLuma( c );
    float mx = max( c.r, max( c.g, c.b ) );
    float mn = min( c.r, min( c.g, c.b ) );
    float sat = ( mx - mn ) / max( mx, 1.0e-3 );
    float glare = smoothstep( 0.55, 1.0, l ) * ( 1.0 - sat );
    c *= 1.0 - 0.5 * glare * amt * gKA * 0.6;
    return mix( vec3( l ), c, 1.0 + 0.5 * amt * gKB * 0.6 );
}

vec4 trans( vec2 p, float d )
{
    float amt = trMid( d ) * ( 1.0 + 0.2 * audioSwell );
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec4  c = mix( vec4( polarise( a.rgb, amt ), a.a ), vec4( polarise( b.rgb, amt ), b.a ), smoothstep( 0.1, 0.9, d ) );
    return c;
}

=== RaindropLensRefraction
brief: TRANSITION RAINDROP LENS REFRACTION: drops on the lens refract the picture, each showing the new scene inverted, and they run together.
about: Elongated drops slide slowly down a pane in front of the picture; each is a little lens that shows the incoming scene upside down and magnified, with a bright edge; they grow and merge until the new scene fills the pane.

---
vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    float asp = resolution.x / resolution.y;
    float n = ( 5.0 + 4.0 * gKA );
    vec2  q = vec2( p.x * asp, p.y - 0.002 * trClock() * 8.0 ) * n;
    vec2  g = floor( q );
    float best = 9.0;
    vec2  bc = g;
    float rnd = 0.0;
    for ( int j = -1; j <= 1; ++j )
        for ( int i = -1; i <= 1; ++i )
        {
            vec2  c = g + vec2( float( i ), float( j ) );
            vec2  pt = c + 0.2 + 0.6 * hash22( c + 3.9 );
            vec2  r = q - pt;
            float dd = length( r * vec2( 1.0, 0.62 ) );
            if ( dd < best )
            {
                best = dd;
                bc = pt;
                rnd = hash21( c + 5.5 );
            }
        }
    float grow = smoothstep( 0.0, 0.85, d );
    float R = ( 0.2 + 0.25 * rnd ) * ( 0.3 + 1.6 * grow );
    float inside = 1.0 - smoothstep( R * 0.9, R, best );
    float upright = smoothstep( 0.55, 1.0, d );
    float k = mix( -0.8 * gKB, 1.0, upright );
    vec2  pc = vec2( bc.x / asp, bc.y + 0.002 * trClock() * 8.0 * n ) / n;
    vec2  uv = pc + ( p - pc ) * k;
    vec4  lens = tBm( uv );
    float rim = smoothstep( R * 0.7, R, best ) * inside;
    lens.rgb += vec3( 0.9, 0.95, 1.0 ) * rim * 0.18 * trMid( d ) * 2.0;
    vec4  c0 = mix( a, lens, inside );
    return mix( c0, tB( p ), smoothstep( 0.82, 1.0, d ) );
}

/**  A caustic network built from three crossing waves.
 *
 * @param q  The place in lattice units. (in)
 * @param t  The clock. (in)
 *
 * @return
 *   The brightness of the lattice, 0..1 (1 on the bright lines).
 **/
float causticNet( vec2 q, float t )
{
    float v = 0.0;
    for ( int i = 0; i < 3; ++i )
    {
        float a = 2.0943951 * float( i ) + 0.4;
        vec2  dir = vec2( cos( a ), sin( a ) );
        float ph = dot( q, dir ) + 0.6 * sin( dot( q, vec2( -dir.y, dir.x ) ) * 0.9 + t * ( 1.0 + 0.3 * float( i ) ) );
        v += cos( ph * 3.0 );
    }
    return pow( clamp( v / 3.0 * 0.5 + 0.5, 0.0, 1.0 ), 3.0 );
}

vec4 trans( vec2 p, float d )
{
    vec4  a = tA( p );
    vec4  b = tB( p );
    vec2  q = p * vec2( resolution.x / resolution.y, 1.0 ) * ( 3.2 * gKA + 3.0 );
    float net = causticNet( q, trClock() * 3.0 * gKB );
    float f = clamp( 0.4 * ( 1.0 - net * 1.5 ) + 0.6 * smoothstep( 0.25, 0.75, fbm2( p * 2.5 ) ), 0.0, 1.0 );
    float w = trThresh( f, d, 0.3 );
    vec4  c = mix( a, b, w );
    c.rgb += vec3( 0.7, 0.9, 1.0 ) * net * 0.18 * trMid( d ) * 2.0 * ( 1.0 + 0.4 * audioSwell );
    return c;
}
